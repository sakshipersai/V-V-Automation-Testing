use JDB;


create table empinfo(
emp_id INT PRIMARY KEY,
emp_name VARCHAR(100),
pid_id INT            //foreign key
);



create table perinfo(
pid_id INT PRIMARY KEY,
pid_name varchar(100) NOT NULL
);

SELECT e.em_name,p.pid_name FROM empinfo e                                      //INNER JOIN 
INNER JOIN perinfo p ON e.pid_id=p.pid_id;                                     


SELECT e.emp_name,p.pid_name FROM empinfo e                                     //LEFT OUTER JOIN
LEFT JOIN perinfo p ON e.pid_id=p.pid_id;


SELECT e.emp_name=p.pid_name FROM empinfo e                                     //RIGHT JOIN
RIGHT JOIN perinfo p ON e.pid_id=p.pid_id;
