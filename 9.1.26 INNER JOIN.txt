CREATE TABLE perinfo(
pid_id INTEGER PRIMARY KEY,
pid_name VARCHAR(50)
);

CREATE TABLE empinfo(
emp_id INTEGER PRIMARY KEY,
emp_name VARCHAR(100),
pid_id INTEGER,
CONSTRAINT fk_pid
FOREIGN KEY(pid_id)
REFERENCES perinfo(pid_id)
);
